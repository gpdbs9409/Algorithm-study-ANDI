# 접근방식
#
# 1. 먼저 50×50 크기의 2차원 배열을 생성한다.
#    모든 셀의 초기값은 "EMPTY"로 설정한다.
#
# 2. commands의 각 문자열을 split()으로 분리한다.
#    parts[0]으로 명령어 종류를 확인하고 분기 처리한다.
#
# 3. UPDATE r c value
#    특정 위치의 값을 변경한다.
#    단, 병합된 셀이라면 해당 그룹의 대표 셀 값을 변경해야 한다.
#
# 4. UPDATE value1 value2
#    표 전체를 순회하면서 value1을 가진 셀을 찾아 value2로 변경한다.
#
# 5. MERGE r1 c1 r2 c2
#    처음에는 두 셀의 값을 동일하게 변경하면 된다고 생각했다.
#    하지만 이후 UPDATE가 발생해도 병합된 셀들은 같은 값을 가져야 한다.
#
#    따라서 단순히 값을 복사하는 것이 아니라,
#    병합된 모든 셀이 하나의 값을 참조하도록 만들어야 한다.
#
#    처음에는 C의 포인터처럼 같은 값을 참조하는 방식을 떠올렸다.
#    Python에서는 별도의 배열을 만들어 각 셀이 대표 셀을 가리키도록
#    구현할 수 있다고 생각했다.
#
#    또는 딕셔너리를 이용해
#    {(1, 1): {(1, 1), (1, 2), (1, 3)}}
#    처럼 그룹을 관리하는 방법도 고민했다.
#
#    하지만 그룹끼리 다시 병합될 수 있기 때문에
#    각 셀이 속한 그룹을 관리하는 방법이 필요했다.
#
#    => Union-Find(서로소 집합) 자료구조를 활용한다.
#
#    Find: 특정 셀이 속한 그룹의 대표 셀을 찾는다.
#    Union: 두 셀이 속한 그룹을 하나로 합친다.
#
#    matrix: 각 대표 셀이 관리하는 값을 저장한다.
#    parent: 각 셀이 어느 대표 셀을 가리키는지 저장한다.
#
#    병합되지 않은 셀은 자기 자신을 대표로 가진다.
#    따라서 별도의 병합 여부 T/F 배열은 필요하지 않다.
#
# 6. UNMERGE
#    아직 구현하지 않음.
#
# 7. PRINT
#    아직 구현하지 않음.

def solution(commands):

    # 50×50 배열 생성 (좌표가 1부터 시작하므로 51×51 사용)
    matrix = [["EMPTY"] * 51 for _ in range(51)]

    # 각 셀이 가리키는 대표 셀을 저장
    parent = [
        [(r, c) for c in range(51)]
        for r in range(51)
    ]

    answer = []

    # 특정 셀이 속한 그룹의 대표 셀 찾기
    def find(r, c):

        if parent[r][c] == (r, c):
            return (r, c)

        pr, pc = parent[r][c]
        parent[r][c] = find(pr, pc)

        return parent[r][c]

    for command in commands:
        parts = command.split()

        # 첫 번째 조건: UPDATE r c value
        if parts[0] == "UPDATE" and len(parts) == 4:

            r = int(parts[1])
            c = int(parts[2])
            value = parts[3]

            # 병합된 셀이라면 대표 셀의 값을 변경
            pr, pc = find(r, c)
            matrix[pr][pc] = value

        # 두 번째 조건: UPDATE value1 value2
        elif parts[0] == "UPDATE" and len(parts) == 3:

            value1 = parts[1]
            value2 = parts[2]

            # 표 전체를 순회하며 값 변경
            for r in range(1, 51):
                for c in range(1, 51):

                    pr, pc = find(r, c)

                    if matrix[pr][pc] == value1:
                        matrix[pr][pc] = value2

        # 세 번째 조건: MERGE r1 c1 r2 c2
        elif parts[0] == "MERGE":

            r1 = int(parts[1])
            c1 = int(parts[2])
            r2 = int(parts[3])
            c2 = int(parts[4])

            # 각각의 대표 셀 찾기
            pr1, pc1 = find(r1, c1)
            pr2, pc2 = find(r2, c2)

            # 이미 같은 그룹이라면 무시
            if (pr1, pc1) == (pr2, pc2):
                continue

            # 병합 후 유지할 값 결정
            value1 = matrix[pr1][pc1]
            value2 = matrix[pr2][pc2]

            if value1 != "EMPTY":
                value = value1
            else:
                value = value2

            # 두 번째 그룹의 대표를 첫 번째 그룹에 연결
            parent[pr2][pc2] = (pr1, pc1)

            # 첫 번째 그룹의 대표 셀에 값 저장
            matrix[pr1][pc1] = value

            # 이전 대표 셀의 값 초기화
            matrix[pr2][pc2] = "EMPTY"

        # 네 번째 조건: UNMERGE r c
        elif parts[0] == "UNMERGE":

            # TODO: 병합 해제 구현
            pass

        # 다섯 번째 조건: PRINT r c
        elif parts[0] == "PRINT":

            # TODO: 셀 값 조회 및 answer에 추가
            pass

    return answer
