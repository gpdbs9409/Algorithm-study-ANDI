처음 생각한 방법과 문제점

처음에는 조건에 맞는 순서(예: 1 2 4 3, 1 2 3 4 …)를 전부 만들어 두고, 순서마다 거리를 더해서 그중 가장 짧은 것을 고르려고 했다.

답은 맞게 나오지만 경우의 수가 너무 많다. 패널이 15개인데 조건이 [1, 2] 하나뿐이면, 조건에 맞는 순서만 해도 약 15! / 2 ≈ 6500억 개다. 그래서 시간 초과가 난다.
처음에는 각 경로별 가능한 순서를 정해두고, 그다음에 가능한 조합을 먼저 다 간뒤에 그 중 최단거리를 구하려고 했다.

그런데 너무 많은 경우의 수로 인해 시간초과가 났다. 그래서 방법을 바꿨다. 1번에서 시작해서 처음에 bfs를 돌려서 각 패널로의 모든 경우를 탐색하고, 2번에서도 시작해서 각 3번, 4번 등등 있는 나머지 부분을 탐색한다. bfs를 층마다 돌릴 필요 없다.

처음에 엘리베이터부터 각 패널의 거리, 1번패널에서 엘리베이터의 거리만 구하면 bfs를 매번 구할 필요 없이 어짜피 엘리베이터를 통해서 다른 층으로 가기 때문에 그러면 끝났다. 심지어 각 층의 구조도 같아서 bfs를 한번씩만 돌리면 끝났다.

BFS를 이용해서 각 패널의 거리를 담을 수 있는 표를 만들었다(이건 모든 층의 구조가 같기 때문에 가낭했었다) 거기에 엘리베이터 타는 것만 고려하면 되었다.

순서도 처음에는 다 기억하려고 했는데 하나하나 다 만들면 너무 많아서 겹치는 경우를 하나로 생각했다.

ex ) 1 2 4 3 | 5 6 7 ...
ex ) 2 1 4 3 | 5 6 7 ...
라면 1 2 3 4 켰고 3번에 서있으니까, 이때는 경우가 똑같고 기억하는 건 더 짧은 것만 기억하면 된다.

### 풀이

dp[mask][last] = mask에 있는 패널들을 켰고, 마지막으로 last를 켠 상태의 최소 시간
mask는 어떤 패널을 켰는지를 비트로 나타낸 것이다. 상태 수가 2¹⁵ × 15 ≈ 50만 개라서 충분히 빠르다.

선행 조건 체크 pre[j] = j를 켜기 전에 켜져 있어야 하는 패널(비트)이다. (pre[j] & mask) == pre[j]이면 필요한 패널이 다 켜져 있다는 뜻이므로 j를 켤 수 있다.

시작 상태 1번 패널에도 선행 조건이 있을 수 있어서 1번을 무조건 먼저 켜면 안 된다. "1번 자리에 서 있고 아직 아무것도 안 켠 상태"에서 출발해서, 선행 조건이 없는 패널 j로 가는 비용 dist[0][j]로 시작한다.

## 코드 (C++)

```cpp
#include <bits/stdc++.h>
using namespace std;

const int INF = 1000000000;

vector<vector<int>> bfs(vector<string>& grid, int startRow, int startCol) {
    int rows = grid.size();
    int cols = grid[0].size();
    vector<vector<int>> distance(rows, vector<int>(cols, INF));

    int moveRow[4] = {-1, 1, 0, 0};
    int moveCol[4] = {0, 0, -1, 1};

    queue<pair<int, int>> q;
    distance[startRow][startCol] = 0;
    q.push({startRow, startCol});

    while (!q.empty()) {
        int row = q.front().first;
        int col = q.front().second;
        q.pop();

        for (int dir = 0; dir < 4; dir++) {
            int nextRow = row + moveRow[dir];
            int nextCol = col + moveCol[dir];

            if (nextRow < 0 || nextRow >= rows || nextCol < 0 || nextCol >= cols) continue;
            if (grid[nextRow][nextCol] == '#') continue;            // 벽
            if (distance[nextRow][nextCol] != INF) continue;       // 이미 방문

            distance[nextRow][nextCol] = distance[row][col] + 1;
            q.push({nextRow, nextCol});
        }
    }
    return distance;
}

int solution(int h, vector<string> grid, vector<vector<int>> panels, vector<vector<int>> seqs) {
    int rows = grid.size();
    int cols = grid[0].size();
    int panelCount = panels.size();

    int elevatorRow = 0, elevatorCol = 0;
    for (int r = 0; r < rows; r++)
        for (int c = 0; c < cols; c++)
            if (grid[r][c] == '@') { elevatorRow = r; elevatorCol = c; }


    vector<int> panelFloor(panelCount), panelRow(panelCount), panelCol(panelCount);
    for (int i = 0; i < panelCount; i++) {
        panelFloor[i] = panels[i][0];
        panelRow[i]   = panels[i][1] - 1;
        panelCol[i]   = panels[i][2] - 1;
    }

    vector<vector<int>> distFromElevator = bfs(grid, elevatorRow, elevatorCol);

    vector<vector<vector<int>>> distFromPanel(panelCount);
    for (int i = 0; i < panelCount; i++)
        distFromPanel[i] = bfs(grid, panelRow[i], panelCol[i]);


    vector<vector<int>> travelTime(panelCount, vector<int>(panelCount, INF));
    for (int from = 0; from < panelCount; from++) {
        for (int to = 0; to < panelCount; to++) {
            if (panelFloor[from] == panelFloor[to]) {
                travelTime[from][to] = distFromPanel[from][panelRow[to]][panelCol[to]];
            } else {
                int toElevator   = distFromElevator[panelRow[from]][panelCol[from]];
                int fromElevator = distFromElevator[panelRow[to]][panelCol[to]];
                int floorGap     = abs(panelFloor[from] - panelFloor[to]);
                if (toElevator < INF && fromElevator < INF)
                    travelTime[from][to] = toElevator + floorGap + fromElevator;
            }
        }
    }
    vector<int> required(panelCount, 0);
    for (auto& seq : seqs) {
        int before = seq[0] - 1;
        int after  = seq[1] - 1;
        required[after] |= (1 << before);
    }

    int allOn = (1 << panelCount) - 1;
    vector<vector<int>> minTime(1 << panelCount, vector<int>(panelCount, INF));

    for (int first = 0; first < panelCount; first++) {
        if (required[first] == 0 && travelTime[0][first] < INF)
            minTime[1 << first][first] = travelTime[0][first];
    }

    for (int onMask = 1; onMask <= allOn; onMask++) {
        for (int lastPanel = 0; lastPanel < panelCount; lastPanel++) {
            int timeSoFar = minTime[onMask][lastPanel];
            if (timeSoFar >= INF) continue;

            for (int nextPanel = 0; nextPanel < panelCount; nextPanel++) {
                if (onMask & (1 << nextPanel)) continue;                               // 이미 켬
                if ((required[nextPanel] & onMask) != required[nextPanel]) continue;   // 선행 조건 미충족
                if (travelTime[lastPanel][nextPanel] >= INF) continue;                 // 갈 수 없음

                int nextMask = onMask | (1 << nextPanel);
                minTime[nextMask][nextPanel] = min(minTime[nextMask][nextPanel],
                                                   timeSoFar + travelTime[lastPanel][nextPanel]);
            }
        }
    }

    //전부 켠 상태 중 최솟값
    int answer = INF;
    for (int lastPanel = 0; lastPanel < panelCount; lastPanel++)
        answer = min(answer, minTime[allOn][lastPanel]);
    return answer;
}
```
