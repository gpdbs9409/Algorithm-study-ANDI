$n$번째 칸에 도달하는 마지막 행동은 결국 '$n-1$번째 칸에서 1칸 뛰기' 또는 '$n-2$번째 칸에서 2칸 뛰기' 두 가지뿐이라, 두 경우의 수를 합친 것

($f(n) = f(n-1) + f(n-2)$)

=> 피보나치 수열과 일치함

```
#include <string>
#include <vector>

using namespace std;

long long solution(int n) {
    if (n == 0) return 0;
    if (n == 1) return 1;
    if( n == 2) return 2;

    int prev2 = 1;
    int prev1 = 2;
    int current = 0;

    for (int i = 3; i <= n; i++) {
        current = (prev1 + prev2) % 1234567; // 앞의 두 숫자를 더하고 나머지 구하기


        prev2 = prev1;
        prev1 = current;
    }

    return current;
}
```
