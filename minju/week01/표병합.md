- 표 병합 (Union-Find)
- Union-Find를 써야 하는 이유
  - 병합(MERGE), 해제(UNMERGE), 값 갱신(UPDATE)이 일어날 때 여러 셀이 하나의 그룹처럼 움직여야 하기 때문.
  - 2차원 배열만으로 안 되는 이유
    - `(1,1)`과 `(1,2)`를 병합했다고 하면, 이후 둘 중 하나의 값을 바꾸거나 UNMERGE 해야 한다.
    - 단순 2차원 배열(`table[1][1]`, `table[1][2]`)은 두 칸이 같은 그룹이라는 정보를 담지 못해 서로 독립된 칸으로만 존재한다.
    - 값이 바뀔 때마다 50x50 전체를 뒤져서 연결 여부를 확인해야 하는 문제가 생긴다.
  - Union-Find로 해결하는 방법
    - 모든 병합된 셀이 대표 좌표(root)를 하나씩 가지도록 만든다.
    - 대표 지정: `(1,1)`과 `(1,2)`를 병합하면 한쪽(`(1,1)`)을 root로 삼고, `(1,2)`는 "내 root는 (1,1)"이라고 기록한다. 이 root를 찾아가는 과정이 `find` 연산이다.
    - 값 저장 위치: 값은 root 칸에만 저장한다. PRINT/UPDATE 시 `find`로 root를 찾은 뒤 root 칸의 값을 읽거나 수정하면, 그룹 내 어떤 칸을 선택해도 같은 값을 공유하게 된다.
  - MERGE에서 Union-Find가 하는 일
    - `(r1,c1)`과 `(r2,c2)` 각각의 root를 찾고(`find`), 두 root가 다르면 한쪽 root가 다른쪽 root를 부모로 가리키게 합친다(`union`).
  - 요약
    - Union-Find는 셀들을 부모-자식 관계로 연결해주는 도구다.
    - 이 구조가 없으면 병합된 셀들이 같이 움직이거나, UNMERGE 시 어떤 셀들이 연결되어 있었는지 추적할 방법이 없다.

- find 구현
  - `find_root`는 재귀로 parent를 계속 타고 올라가며, parent 좌표가 현재 좌표와 같아질 때(= 자기 자신이 root일 때)까지 거슬러 올라간다.

- 1차 코드 (MERGE/UNMERGE 미구현)

  ```cpp
  #include <bits/stdc++.h>
  using namespace std;

  pair<int, int> find_root(int r, int c, vector<vector<pair<int,int>>>& parent){
      if(parent[r][c].first == r && parent[r][c].second == c){
          return {r,c};
      }
      return parent[r][c] = find_root(parent[r][c].first, parent[r][c].second, parent);
  }

  vector<string> solution(vector<string> commands) {
      vector<string> answer;
      vector<vector<string>> table(51, vector<string>(51, ""));
      vector<vector<pair<int,int>>> parent(51, vector<pair<int, int>>(51));
      for(int r=1; r<=50; r++){
          for(int c=1; c<=50; c++){
              parent[r][c] = {r,c};
          }
      }
      for(int i=0; i<commands.size(); i++){
          string command = commands[i];
          stringstream ss(command);
          string op;
          ss >> op;

          if(op == "UPDATE"){
              string first, second, third;
              ss >> first >> second;
              if(ss >> third){
                  int r = stoi(first);
                  int c = stoi(second);
                  table[r][c] = third;
              } else {
                  string value1 = first, value2 = second;
                  for(int r=1; r<=50; r++){
                      for(int c=1; c<=50; c++){
                          if(table[r][c] == value1){
                              table[r][c] = value2;
                          }
                      }
                  }
              }
          }
          else if(op == "MERGE"){
              string r1, c1, r2, c2;
              ss >> r1 >> c1 >> r2 >> c2;
          }
          else if(op == "UNMERGE"){
              string r, c;
              ss >> r >> c;
          }
          else if(op == "PRINT"){
              string rr, cc;
              ss >> rr >> cc;
              int r = stoi(rr), c = stoi(cc);
              answer.push_back(table[r][c] == "" ? "EMPTY" : table[r][c]);
          }
      }
      return answer;
  }
  ```

  - MERGE까지는 이해했고, UNMERGE가 남은 상태.
  - MERGE/UNMERGE는 혼자 짜지 못해 도움을 받았는데, UPDATE/초기화/PRINT도 결국 root를 거쳐야 해서 전체 코드를 다시 수정해야 했음.

- 최종 코드

  ```cpp
  #include <bits/stdc++.h>
  using namespace std;

  pair<int, int> find_root(int r, int c, vector<vector<pair<int,int>>>& parent){
      if(parent[r][c].first == r && parent[r][c].second == c){
          return {r,c};
      }
      return parent[r][c] = find_root(parent[r][c].first, parent[r][c].second, parent);
  }

  vector<string> solution(vector<string> commands) {
      vector<string> answer;
      vector<vector<string>> table(51, vector<string>(51, ""));
      vector<vector<pair<int,int>>> parent(51, vector<pair<int, int>>(51));
      for(int r=1; r<=50; r++){
          for(int c=1; c<=50; c++){
              parent[r][c] = {r,c};
          }
      }

      for(int i=0; i<commands.size(); i++){
          string command = commands[i];
          stringstream ss(command);
          string op;
          ss >> op;

          if(op == "UPDATE"){
              string first, second, third;
              ss >> first >> second;
              if(ss >> third){
                  int r = stoi(first), c = stoi(second);
                  pair<int, int> root = find_root(r, c, parent);
                  table[root.first][root.second] = third;
              } else {
                  string value1 = first, value2 = second;
                  for(int r=1; r<=50; r++){
                      for(int c=1; c<=50; c++){
                          pair<int, int> root = find_root(r, c, parent);
                          if(table[root.first][root.second] == value1){
                              table[root.first][root.second] = value2;
                          }
                      }
                  }
              }
          }
          else if(op == "MERGE"){
              string rr1, cc1, rr2, cc2;
              ss >> rr1 >> cc1 >> rr2 >> cc2;
              int r1 = stoi(rr1), c1 = stoi(cc1);
              int r2 = stoi(rr2), c2 = stoi(cc2);

              pair<int, int> root1 = find_root(r1, c1, parent);
              pair<int, int> root2 = find_root(r2, c2, parent);
              if(root1 == root2) continue;

              parent[root2.first][root2.second] = root1;

              if(table[root1.first][root1.second] == "" && table[root2.first][root2.second] != ""){
                  table[root1.first][root1.second] = table[root2.first][root2.second];
              }
              table[root2.first][root2.second] = "";
          }
          else if(op == "UNMERGE"){
              string rr, cc;
              ss >> rr >> cc;
              int r = stoi(rr), c = stoi(cc);

              pair<int, int> root = find_root(r, c, parent);
              string saved_value = table[root.first][root.second];

              vector<pair<int, int>> group_members;
              for(int i_r=1; i_r<=50; i_r++){
                  for(int i_c=1; i_c<=50; i_c++){
                      if(find_root(i_r, i_c, parent) == root){
                          group_members.push_back({i_r, i_c});
                      }
                  }
              }

              for(auto cell : group_members){
                  parent[cell.first][cell.second] = {cell.first, cell.second};
                  table[cell.first][cell.second] = "";
              }

              table[r][c] = saved_value;
          }
          else if(op == "PRINT"){
              string rr, cc;
              ss >> rr >> cc;
              int r = stoi(rr), c = stoi(cc);
              pair<int, int> root = find_root(r, c, parent);
              answer.push_back(table[root.first][root.second] == "" ? "EMPTY" : table[root.first][root.second]);
          }
      }
      return answer;
  }
  ```

- UNMERGE 설계 (5단계)
  1. root 찾기: `(r, c)`가 속한 그룹의 root를 찾는다. `find_root(r, c, parent)`
  2. 값 백업: root 칸에 있던 값을 변수에 복사해 둔다. 문제 조건상 해제 전 값이 있었다면 `(r, c)`가 그 값을 가져야 하므로 미리 저장. `table[root.first][root.second]`
  3. 그룹원 전부 찾기: 50x50 전체를 돌며 `find_root(i_r, i_c, parent) == root`인 칸을 모두 `group_members`에 담는다.
  4. 그룹원 초기화: `group_members`의 각 셀을 순회하며 parent를 자기 자신으로, table 값을 `""`으로 되돌린다.
  5. 값 복원: `(r, c)`에만 2단계에서 백업한 값을 넣는다.

- 설계 프로세스
  - Phase 1. 자료구조 설계
    1. 표(Grid): 50x50이므로 `vector<vector<string>> table(51, vector<string>(51, ""))`
    2. 부모 정보: 셀 간 연결 관계를 담는 `vector<vector<pair<int,int>>> parent(51, vector<pair<int,int>>(51))`. 초기에는 모든 칸이 자기 자신을 root로 가지므로 이중 for문으로 `parent[r][c] = {r, c}` 초기화.
    3. find_root 함수: parent가 자기 자신이 될 때까지 재귀로 올라가는 함수를 먼저 작성.
  - Phase 2. 명령어 파싱
    - 명령어가 `"UPDATE 1 1 menu"` 형태의 문자열로 들어오므로 `stringstream`으로 분리한다.

      ```cpp
      stringstream ss(command);
      string op;
      ss >> op;
      ```

    - 첫 단어(`op`)로 `if (op == "UPDATE")`, `else if (op == "MERGE")` 등 분기를 만든다.
  - Phase 3. 명령어별 로직
    - UPDATE
      - 좌표가 주어지면(`r c value`): `find_root`로 root를 찾아 root의 `table` 값을 바꾼다.
      - 값이 주어지면(`value1 value2`): 전체를 돌며 각 칸의 root를 찾고, root의 값이 `value1`이면 `value2`로 바꾼다.
    - MERGE
      - `root1`, `root2`를 찾는다. 같으면 아무 것도 하지 않는다.
      - 다르면 `parent[root2] = root1`로 합친다.
      - 값 규칙: 한쪽만 값이 있으면 `root1`에 몰아주고 나머지는 비운다.
    - UNMERGE
      - 위 5단계 순서를 그대로 따른다: root 찾기 → 값 백업 → 그룹원 색출 → 그룹원 초기화 → `(r, c)`에 값 복원.
    - PRINT
      - `find_root`로 root를 찾고, root의 `table` 값이 비어 있으면 `"EMPTY"`, 아니면 그 값을 `answer`에 추가한다.
  - 다음에 적용할 접근
    - 비슷한 문제를 만나면 코드를 바로 치지 말고 다음 세 가지를 먼저 정리한다.
    1. 상태를 저장할 자료구조는 무엇인가 (`table`, `parent`)
    2. 이 문제의 핵심 규칙은 무엇인가 (병합된 셀은 root를 공유한다)
    3. 각 명령어별로 어떤 순서로 처리해야 하는가 (Phase 3 시뮬레이션)

- 프로세스
  - Step 1. 기본 뼈대와 find_root 함수
    - 표는 50x50이므로 문자열을 담을 `table`과 부모 좌표를 담을 `parent` 배열이 필요하다.
    - `parent`는 초기에 모든 칸이 자기 자신을 root로 가지므로 `parent[r][c] = {r, c}`로 초기화한다.
    - root를 찾아가는 `find_root` 함수를 가장 먼저 작성한다.

      ```cpp
      #include <bits/stdc++.h>
      using namespace std;

      // 경로 압축을 적용한 root 탐색 함수
      pair<int, int> find_root(int r, int c, vector<vector<pair<int,int>>>& parent){
          if(parent[r][c].first == r && parent[r][c].second == c){
              return {r,c};
          }
          return parent[r][c] = find_root(parent[r][c].first, parent[r][c].second, parent);
      }

      vector<string> solution(vector<string> commands){
          vector<string> answer;

          vector<vector<string>> table(51, vector<string>(51, ""));
          vector<vector<pair<int,int>>> parent(51, vector<pair<int, int>>(51));

          for(int r = 1; r <= 50; r++){
              for(int c = 1; c <= 50; c++){
                  parent[r][c] = {r, c};
              }
          }
      ```

  - Step 2. 명령어 파싱 틀 잡기
    - 명령어가 한 줄씩 들어오므로 `stringstream`으로 첫 단어(`op`)를 뽑아 종류별로 분기한다.

      ```cpp
          for(int i = 0; i < commands.size(); i++){
              string command = commands[i];
              stringstream ss(command);
              string op;
              ss >> op;

              if(op == "UPDATE"){
                  // UPDATE 처리
              }
              else if(op == "MERGE"){
                  // MERGE 처리
              }
              else if(op == "UNMERGE"){
                  // UNMERGE 처리
              }
              else if(op == "PRINT"){
                  // PRINT 처리
              }
          }
          return answer;
      }
      ```

  - Step 3. UPDATE
    - 인자가 3개면(`r c value`) 특정 칸을 바꾸고, 2개면(`value1 value2`) 표 전체를 돌며 바꾼다.
    - 병합되어 있을 수 있으므로 반드시 `find_root`로 root를 찾아 그 root의 `table` 값을 바꿔야 한다.

      ```cpp
      if(op == "UPDATE"){
          string first, second, third;
          ss >> first >> second;

          if(ss >> third){
              // UPDATE r c value
              int r = stoi(first);
              int c = stoi(second);
              string value = third;

              pair<int, int> root = find_root(r, c, parent);
              table[root.first][root.second] = value;
          } else {
              // UPDATE value1 value2
              string value1 = first;
              string value2 = second;

              for(int r = 1; r <= 50; r++){
                  for(int c = 1; c <= 50; c++){
                      pair<int, int> root = find_root(r, c, parent);
                      if(table[root.first][root.second] == value1){
                          table[root.first][root.second] = value2;
                      }
                  }
              }
          }
      }
      ```

  - Step 4. MERGE
    - 두 좌표의 root(`root1`, `root2`)를 각각 찾는다. 이미 같은 root면 할 일이 없으므로 `continue`.
    - 다르면 `root2`의 부모를 `root1`로 설정해 합친다.
    - 값 규칙: 한쪽만 값이 있으면 `root1`에 몰아주고 나머지는 비운다.

      ```cpp
      else if(op == "MERGE"){
          string rr1, cc1, rr2, cc2;
          ss >> rr1 >> cc1 >> rr2 >> cc2;
          int r1 = stoi(rr1), c1 = stoi(cc1);
          int r2 = stoi(rr2), c2 = stoi(cc2);

          pair<int, int> root1 = find_root(r1, c1, parent);
          pair<int, int> root2 = find_root(r2, c2, parent);

          if(root1 == root2) continue; // 이미 같은 그룹

          parent[root2.first][root2.second] = root1;

          if(table[root1.first][root1.second] == "" && table[root2.first][root2.second] != ""){
              table[root1.first][root1.second] = table[root2.first][root2.second];
          }
          table[root2.first][root2.second] = "";
      }
      ```

  - Step 5. UNMERGE
    1. 현재 칸의 root를 찾는다 (`root`)
    2. root의 값을 백업한다 (`saved_value`)
    3. 표 전체를 돌며 root가 같은 모든 칸을 찾는다 (`group_members`)
    4. 그 칸들의 parent를 자기 자신으로 되돌리고 값도 지운다
    5. `(r, c)`에만 백업한 값을 넣는다

      ```cpp
      else if(op == "UNMERGE"){
          string rr, cc;
          ss >> rr >> cc;
          int r = stoi(rr), c = stoi(cc);

          pair<int, int> root = find_root(r, c, parent);
          string saved_value = table[root.first][root.second];

          vector<pair<int, int>> group_members;
          for(int i_r = 1; i_r <= 50; i_r++){
              for(int i_c = 1; i_c <= 50; i_c++){
                  if(find_root(i_r, i_c, parent) == root){
                      group_members.push_back({i_r, i_c});
                  }
              }
          }

          for(auto cell : group_members){
              parent[cell.first][cell.second] = {cell.first, cell.second};
              table[cell.first][cell.second] = "";
          }

          table[r][c] = saved_value;
      }
      ```

  - Step 6. PRINT
    - 출력할 때도 내 칸을 바로 보지 않고 root를 찾아 그 root의 `table` 값을 확인한다.
    - 비어있으면 `"EMPTY"`, 아니면 그 값을 `answer`에 추가한다.

      ```cpp
      else if(op == "PRINT"){
          string rr, cc;
          ss >> rr >> cc;
          int r = stoi(rr), c = stoi(cc);

          pair<int, int> root = find_root(r, c, parent);

          if(table[root.first][root.second] == ""){
              answer.push_back("EMPTY");
          } else {
              answer.push_back(table[root.first][root.second]);
          }
      }
      ```
