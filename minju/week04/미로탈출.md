# 미로 탈출 (Lv2)

레버까지 bfs, 그 이후에 출구까지 bfs 하면됨

- 코드

  ```jsx
  #include <string>
  #include <vector>
  #include <queue>

  using namespace std;

  int dx[4] = {0, 0, -1, 1};
  int dy[4] = {-1, 1, 0, 0};

  // 문자에서 문자까지의 최단거리
  int bfs(char start_char, char target_char, const vector<string>& maps) {
      int n = maps.size();
      int m = maps[0].size();
      int start_x = -1, start_y = -1;

      for (int i = 0; i < n; i++) {
          for (int j = 0; j < m; j++) {
              if (maps[i][j] == start_char) {
                  start_x = i;
                  start_y = j;
                  break;
              }
          }
          if (start_x != -1) break;
      }

      vector<vector<int>> dist(n, vector<int>(m, -1));
      queue<pair<int, int>> q;

      q.push({start_x, start_y});
      dist[start_x][start_y] = 0;

      while (!q.empty()) {
          int x = q.front().first;
          int y = q.front().second;
          q.pop();

          if (maps[x][y] == target_char) {
              return dist[x][y];
          }

          for (int i = 0; i < 4; i++) {
              int nx = x + dx[i];
              int ny = y + dy[i];

              // 범위체크
              if (nx < 0 || nx >= n || ny < 0 || ny >= m) continue;
              if (maps[nx][ny] == 'X') continue;


              if (dist[nx][ny] == -1) {
                  dist[nx][ny] = dist[x][y] + 1;
                  q.push({nx, ny});
              }
          }
      }

      // 도달하지 못했다면 -1 리턴
      return -1;
  }

  int solution(vector<string> maps) {
      int time1 = bfs('S', 'L', maps);
      if (time1 == -1) return -1;
      int time2 = bfs('L', 'E', maps);
      if (time2 == -1) return -1;
      return time1 + time2;
  }
  ```
