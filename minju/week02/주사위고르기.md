    ```jsx
    #include <bits/stdc++.h>
    using namespace std;
    
    // 주사위 합 배열을 구하는 함수
    void makeSum(int idx, int cSum, const vector<int> &dice_indices, const vector<vector<int>> &dice, vector<int> &sum_list) {
        if (idx == dice_indices.size()) {
            sum_list.push_back(cSum);
            return;
        }
        
        int dice_idx = dice_indices[idx];
        for (int i = 0; i < 6; i++) {
            makeSum(idx + 1, cSum + dice[dice_idx][i], dice_indices, dice, sum_list);
        }
    }
    
    vector<int> solution(vector<vector<int>> dice) {
        int n = dice.size();
        int half = n / 2;
        
        vector<int> choice(n, 0);
        for(int i = half; i < n; i++){
            choice[i] = 1;
        }
        
        int max_wins = -1;
        double max_probability = 0.0;
        vector<int> best_a;
        
        do{
            vector<int> a_dice;
            vector<int> b_dice;
            
            for(int i = 0; i < n; i++){
                if(choice[i] == 0){
                    a_dice.push_back(i);
                } else {
                    b_dice.push_back(i);
                }
            }
               
            vector<int> a_sums;
            vector<int> b_sums;
            makeSum(0, 0, a_dice, dice, a_sums);
            makeSum(0, 0, b_dice, dice, b_sums);
            
            sort(a_sums.begin(), a_sums.end());
            sort(b_sums.begin(), b_sums.end());
            
            // 투 포인터 이용
            int wins = 0;
            int j = 0;
            int b_size = b_sums.size();
            for(int a_val : a_sums){
                while(j < b_size && b_sums[j] < a_val){
                    j++;
                }
                wins += j;
            }
            
            // [추가] 승리 확률(%) 계산하기
            double total_cases = (double)a_sums.size() * b_sums.size();
            double current_probability = ((double)wins / total_cases) * 100.0;
            
            cout << "A : [ ";
            for(int idx : a_dice) cout << (idx + 1) << " ";
            cout << "] | B : [ ";
            for(int idx : b_dice) cout << (idx + 1) << " ";
            cout << "] | 승리 횟수: " << wins << " | 확률: " << fixed << setprecision(2) << current_probability << "%\n";
      
            // 최다 승리 횟수 갱신
            if(wins > max_wins){
                max_wins = wins;
                max_probability = current_probability;
                best_a = a_dice;
            }
            
        } while(next_permutation(choice.begin(), choice.end()));
        
        // 최종 결과 출력
        cout << "마지막 승리 확률: " << max_probability << "%\n";
        
        vector<int> answer;
        for(int idx : best_a){
            answer.push_back(idx + 1);
        }
        sort(answer.begin(), answer.end());
        
        return answer;
    }
    ```

    ```jsx
    #include <string>
    #include <vector>
    #include <algorithm>
    
    using namespace std;
    
    // 선택된 주사위들의 모든 합 경우의 수를 구하는 DFS
    void get_sums(int depth, int current_sum, const vector<int>& selected_dice, const vector<vector<int>>& dice, vector<int>& sums) {
        if (depth == selected_dice.size()) {
            sums.push_back(current_sum);
            return;
        }
        
        int dice_idx = selected_dice[depth];
        for (int i = 0; i < 6; ++i) {
            get_sums(depth + 1, current_sum + dice[dice_idx][i], selected_dice, dice, sums);
        }
    }
    
    vector<int> solution(vector<vector<int>> dice) {
        int n = dice.size();
        
        // next_permutation을 위한 조합 마스크 생성 (0: A가 선택, 1: B가 선택)
        vector<int> mask(n, 0);
        fill(mask.begin() + (n / 2), mask.end(), 1);
        
        int max_wins = -1;
        vector<int> best_a_dice;
        
        do {
            vector<int> a_dice, b_dice;
            for (int i = 0; i < n; ++i) {
                if (mask[i] == 0) a_dice.push_back(i);
                else b_dice.push_back(i);
            }
            
            // 1. A와 B의 모든 합 구하기
            vector<int> a_sums, b_sums;
            get_sums(0, 0, a_dice, dice, a_sums);
            get_sums(0, 0, b_dice, dice, b_sums);
            
            // 2. B의 합배열 정렬 (이분 탐색 / lower_bound 사용을 위해)
            sort(b_sums.begin(), b_sums.end());
            
            // 3. A가 승리하는 횟수 카운트
            int wins = 0;
            for (int a_val : a_sums) {
                // a_val보다 작은 b_val의 개수를 구함
                auto it = lower_bound(b_sums.begin(), b_sums.end(), a_val);
                wins += distance(b_sums.begin(), it);
            }
            
            // 4. 최다 승리 조합 갱신
            if (wins > max_wins) {
                max_wins = wins;
                best_a_dice = a_dice;
            }
            
        } while (next_permutation(mask.begin(), mask.end()));
        
        // 주사위 번호를 1-based 인덱스로 변환 및 정렬
        vector<int> answer;
        for (int idx : best_a_dice) {
            answer.push_back(idx + 1);
        }
        sort(answer.begin(), answer.end());
        
        return answer;
    }
    ```
